// title
title Manuzam ER Diagram

// define tables
users [icon: user, color: yellow] {
  id integer pk
  username string
  email string
  password_hash string
  created_at timestamp
}

machines [icon: cpu, color: blue] {
  id integer pk
  user_id integer
  name string
  manufacturer string
  model_number string
  year_of_installation integer
  operating_hours_per_day integer
  maintenance_schedule string
  last_maintenance_date date
  expected_lifespan integer
  created_at timestamp
}

sensors [icon: activity, color: orange] {
  id integer pk
  machine_id integer
  temperature_sensor_enabled boolean
  vibration_sensor_enabled boolean
}

normal_audio_samples [icon: music, color: green] {
  id integer pk
  machine_id integer unique
  user_id integer
  file_path string
  uploaded_at timestamp
}

faults [icon: alert-triangle, color: red] {
  id integer pk
  machine_id integer
  name string
  description string
}

fault_audio_samples [icon: music, color: purple] {
  id integer pk
  fault_id integer
  user_id integer
  file_path string
  uploaded_at timestamp
}

fault_annotations [icon: edit-3, color: pink] {
  id integer pk
  fault_audio_sample_id integer
  user_id integer
  cause string
  confirmed boolean
  annotated_at timestamp
}

ml_models [icon: cpu, color: teal] {
  id integer pk
  machine_id integer
  model_name string
  model_file_path string
  created_by integer
  created_at timestamp
}

// define relationships
machines.user_id > users.id
sensors.machine_id > machines.id
normal_audio_samples.machine_id > machines.id
normal_audio_samples.user_id > users.id
faults.machine_id > machines.id
fault_audio_samples.fault_id > faults.id
fault_audio_samples.user_id > users.id
fault_annotations.fault_audio_sample_id > fault_audio_samples.id
fault_annotations.user_id > users.id
ml_models.machine_id > machines.id
ml_models.created_by > users.id
